  CLIMBING_UP,

  AT_BOTTOM,

};

RobotState state = LINE_FOLLOW;

  analogWrite(enr, 0);
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void rotate360() {
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  unsigned long now = millis();
  float dt = (now - lastTime) / 1000.0;  // seconds
  lastTime = now;

  yawAngle -= g.gyro.x * dt * 180 / PI; // rad/s -> degrees

  // Start spinning motors
  if (!spinning) {
    spinning = true;
    yawAngle = 0.0;

    digitalWrite(IN1, HIGH);
    digitalWrite(IN2, LOW);
    digitalWrite(IN3, LOW);
    digitalWrite(IN4, HIGH);

    analogWrite(enl, 200);
    analogWrite(enr, 200);
  }

  Serial.print("Yaw: ");
  Serial.println(yawAngle);

  // Stop when 360° reached
  if (abs(yawAngle) >= 184) {
    stopMotors();
    spinning = false;
  }
}

void setup() {
  Serial.begin(9600);
  Wire.begin();

  lcd.begin(16, 2);
  lcd.clear();
  lcd.print("Initializing MPU");

  if (!mpu.begin()) {
    lcd.clear();
    lcd.print("MPU ERROR!");
    while (1);
  }

  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);

  delay(500);
  lcd.clear();
  lcd.print("Ready");

  lastTime = millis();
}

void loop() {
  // MPU UPDATE
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  float dt = (millis() - lastTime) / 1000.0;
  lastTime = millis();

  float accPitch = atan2(a.acceleration.z, a.acceleration.x) * 180 / PI;
  float gyroRatePitch = g.gyro.y * 180 / PI;

  pitch = 0.98 * (pitch + gyroRatePitch * dt) + 0.02 * accPitch;
  //lastTime = millis();

  // Debug
  Serial.print("Pitch: ");
  Serial.println(pitch);

  // STATE MACHINE 

  switch (state) {

    // Normal line following
    case LINE_FOLLOW:
      lineFollow();
      if (pitch > 10.0) {  // Ramp start detected (> 10 degrees positive pitch)
        state = CLIMBING_UP;
        climbingMode = true;
        lcd.setCursor(0, 1);
        lcd.print("CLIMBING_UP  ");
      }
      break;

    // Climbing up ramp and stopping at peak
    case CLIMBING_UP:
      // CRITICAL FIX: Only line follow until the peak is secured
      if (!peakStopped) {
        lineFollow();

        if (pitch > maxPitchReached) {
          maxPitchReached = pitch;
        }
      }

      if (abs(pitch) < 5.0) {  // Peak detected (within 5 degrees of horizontal)
        stopMotors();

        if (!peakStopped) {
          peakStartTime = millis();
          lcd.clear();
          lcd.print("PEAK DETECTED");
          lcd.setCursor(0, 1);
          lcd.print("Angle: ");
          lcd.print(maxPitchReached, 1);
          peakStopped = true; // Lock into the waiting phase
        }

        // Wait for the required duration
        if (millis() - peakStartTime >= 4000) {
          spinning = false;  // Ensure rotation flag is reset
          state = ROTATING;
          lcd.clear();
          lcd.print("State: ROTATING");
          peakStopped = false; // Reset for next cycle (if applicable)
        }
      }
      break;

    // Rotate 360°
    case ROTATING:
      rotate360();
      if (!spinning) {  // Rotation finished
        state = MOVING_DOWN;
        lcd.setCursor(0, 1);
        lcd.print("MOVING_DOWN  ");
      }
      break;

    // Moving down ramp
    case MOVING_DOWN:
      lineFollow();
      if (pitch < -5.0) {  // Bottom detected (small negative pitch suggests level ground)
        stopMotors();
        delay(1000); // Pause for 1 second at the bottom
        state = FINISH;
        lcd.clear();
        lcd.print("State: FINISH");
      }
      break;

    // Continue line following until end (or next condition)
    case FINISH:
      lineFollow();
      break;
  }
}

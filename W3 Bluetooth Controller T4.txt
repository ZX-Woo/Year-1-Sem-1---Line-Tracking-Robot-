#include <LiquidCrystal.h>
#include <SoftwareSerial.h>

// LCD connected to pins RS=8, E=9, D4=4, D5=5, D6=6, D7=7
LiquidCrystal lcd(8, 9, 4, 5, 6, 7); 

// SoftwareSerial(TX, RX)
SoftwareSerial BTSerial(A4, A5); 

const long BLUETOOTH_BAUD_RATE = 9600; 

// -------------------- MOTOR DRIVER PINS --------------------
int IN1 = 13;   // Left motor direction pin 1
int IN2 = 12;   // Left motor direction pin 2
int IN3 = 2;   // Right motor direction pin 1 
int IN4 = A3;    // Right motor direction pin 2 

// -------------------- MOTOR ENABLE PINS --------------------
int enl = 11;   // Left motor enable (PWM)
int enr = 3;    // Right motor enable (PWM)
int default_speed = 120; // Default speed for turning
int tspeed = 180;    // Top speed for straight movement

// SETUP FUNCTION
void setup() {
  // Serial.begin() activates Pin 1 (TX) for PC debugging
  Serial.begin(9600); 
  BTSerial.begin(BLUETOOTH_BAUD_RATE); 
  
  lcd.begin(16, 2);   // Initialize LCD with 16 columns and 2 rows
  lcd.print("Robot Car Ready");
  lcd.setCursor(0, 1);
  lcd.print("CMD: STOP");
  
  // Motor driver pins
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  // Motor enable pins
  pinMode(enl, OUTPUT);
  pinMode(enr, OUTPUT);
  
  // Set initial state to stop
  stopRobot();
  
  // Optional: Send confirmation message over Bluetooth
  BTSerial.println("Robot control initialized. Send FSFS, BSBS, LS, RS, or S (Stop).");
}

// MOTOR CONTROL FUNCTIONS
void stopRobot() {
  digitalWrite(IN1, LOW); 
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
  analogWrite(enl, 0); // Speed 0
  analogWrite(enr, 0); // Speed 0
  lcd.setCursor(0, 1);
  lcd.print("CMD: STOP       ");
}

void moveForward() {
  // Left Motor (HIGH, LOW)
  digitalWrite(IN1, HIGH);  
  digitalWrite(IN2, LOW);
  // Right Motor (HIGH, LOW) 
  digitalWrite(IN3, HIGH);  
  digitalWrite(IN4, LOW);
  analogWrite(enl, tspeed);
  analogWrite(enr, tspeed);
  lcd.setCursor(0, 1);
  lcd.print("CMD: FORWARD    ");
}

void moveBackward() {
  // Left Motor (LOW, HIGH)
  digitalWrite(IN1, LOW);   
  digitalWrite(IN2, HIGH);
  // Right Motor (LOW, HIGH) 
  digitalWrite(IN3, LOW);   
  digitalWrite(IN4, HIGH);
  analogWrite(enl, tspeed);
  analogWrite(enr, tspeed);
  lcd.setCursor(0, 1);
  lcd.print("CMD: BACKWARD   ");
}

void turnLeft() {
  digitalWrite(IN1, LOW);   
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH);  
  digitalWrite(IN4, LOW);
  analogWrite(enl, 220); 
  analogWrite(enr, 200); 
  lcd.setCursor(0, 1);
  lcd.print("CMD: SHARP LEFT   ");
}

void turnRight() {
  digitalWrite(IN1, HIGH);  
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);   
  digitalWrite(IN4, HIGH);
  analogWrite(enl, 200); 
  analogWrite(enr, 220); 
  lcd.setCursor(0, 1);
  lcd.print("CMD: SHARP RIGHT  ");
}

// MAIN LOOP
void loop() {

  if (BTSerial.available()) {
    
    // Read the entire string until the Line Feed (\n) character is received.
    String data = BTSerial.readStringUntil('\n');

    // Remove any leading/trailing whitespace, including the Carriage Return (\r)
    data.trim();
    data.toUpperCase(); // Ensure consistency

    // Only process the command if it's not empty
    if (data.length() > 0) 
    {
      // Print command to serial monitor for debugging
      Serial.print("Received Command: ");
      Serial.println(data);

      if (data.equals("F")) {
        moveForward();
      }
      else if (data.equals("B")) {
        moveBackward();
      }
      else if (data.equals("L")) {
        turnLeft();
      }
      else if(data.equals("R")) {
        turnRight();
      }
      // NEW: Explicitly check for the STOP command.
      else if(data.equals("S")) {
        stopRobot();
      }
      
      else {
        // If the command is unrecognized, treat it as a stop or idle state
        stopRobot();
      }
    }
  }
}
